# Function to create OVN BGP topology components for a given chassis
#
# Args:
#   $1: The chassis name to bind the logical router port to.
#
# Creates:
#   - Logical Router "bgp-router" (if it doesn't exist)
#   - Logical Switch "bgp-switch-<chassis>" (if it doesn't exist)
#   - Logical Router Port on "bgp-router"
#   - Logical Switch Port on "bgp-switch-<chassis>"
#   - Peers the LRP and LSP
#   - Binds the LRP to the specified chassis
#   - Creates a localnet port on "bgp-switch-<chassis>"
#
# All port creation, peering, and binding happens in a single ovn-nbctl transaction.
# Note: This uses example IP/MAC addresses. In a real deployment, you'd need
# a strategy for unique addressing per chassis/segment.
# The localnet port uses 'physnet1' as the network name; adjust if needed.
#
create_ovn_topology_for_chassis() {
    local chassis="$1"
    local physnet_name="$2"
    if [ -z "$chassis" ]; then
        echo "Error: No chassis name provided." >&2
        echo "Usage: create_ovn_topology_for_chassis <chassis_name>" >&2
        return 1
    fi

    local router_name="bgp-router"
    local switch_name="bgp-switch-$chassis"
    local lrp_name="bgp-lrp-${chassis}"
    local lsp_peer_name="bgp-lsp-${chassis}"
    local lsp_localnet_name="lsp-localnet-${chassis}"

    # --- Ensure base resources exist (create outside transaction if needed) ---

    if ! ovn-nbctl list logical_router $router_name > /dev/null; then
        echo "Creating a router"
        if ! ovn-nbctl lr-add "$router_name"; then
            echo "Error creating router '$router_name'." >&2
            return 1
        fi
    fi

    echo "Ensuring switch '$switch_name' exists..."
    if ! ovn-nbctl ls-add "$switch_name"; then
        echo "Error creating switch '$switch_name'." >&2
        return 1
    fi

    # --- Create ports, peer, and bind in a single transaction ---

    # Create an HA chassis group for the chassis
    ovn-nbctl ha-chassis-group-add bgp-$chassis-group \
         -- ha-chassis-group-add-chassis bgp-$chassis-group $chassis 10
    ha_chassis_group=$(ovn-nbctl --column=_uuid --no-headings list ha_chassis_group bgp-$chassis-group)

    echo "Creating ports and peerings in a single transaction for chassis '$chassis'..."

    # Using example IPs 192.168.1.1/24 (LRP) and 192.168.1.2/24 (LSP).
    # Using example MACs 0A:00:00:00:01:01 (LRP) and 0A:00:00:00:01:02 (LSP).
    # You MUST ensure IPs/MACs are unique and appropriate for your network.
    if ovn-nbctl -- \
        lrp-add "$router_name" "$lrp_name" "0A:00:00:00:01:01" "192.168.1.1/24" -- \
        set logical_router_port $lrp_name ha_chassis_group=$ha_chassis_group -- \
        lsp-add "$switch_name" "$lsp_peer_name" -- \
        lsp-set-addresses "$lsp_peer_name" router -- \
        set Logical_Router_Port "$lrp_name" peer="$lsp_peer_name" -- \
        set Logical_Switch_Port "$lsp_peer_name" peer="$lrp_name" -- \
        lsp-add "$switch_name" "$lsp_localnet_name" -- set Logical_Switch_Port "$lsp_localnet_name" type=localnet options:network_name="$physnet_name"; then

        echo "Topology resources created successfully in a single transaction for chassis '$chassis'."
        echo "  - Logical Router Port: $lrp_name (on $router_name, bound to $chassis, Peer: $lsp_peer_name)"
        echo "  - Logical Switch Port: $lsp_peer_name (on $switch_name, Peer: $lrp_name)"
        echo "  - Logical Switch Port: $lsp_localnet_name (on $switch_name, type=localnet, network_name=$physnet_name)"
    else
        echo "Error creating topology resources in a single transaction for chassis '$chassis'." >&2
        return 1
    fi

    return 0
}

function connect_router_and_switch() {
    local router=$1
    local switch=$2
    local mac=$3
    local networks=$4

    local lsp_name=lsp-$switch-to-$router
    local lrp_name=lrp-$router-to-$switch

    ovn-nbctl lrp-add $router $lrp_name $mac "$network" \
           -- lsp-add $switch $lsp_name \
           -- lsp-set-addresses $lsp_name router \
           -- lsp-set-type $lsp_name router \
           -- lsp-set-options $lsp_name router-port=$lrp_name
}
